+// QR Code version 3-L byte-mode encoder. Payloads are short ticket IDs.
+function qrMatrix(text) {
+  const bytes = new TextEncoder().encode(text).slice(0, 53), data = []
+  const put = (value, length) => { for (let i = length - 1; i >= 0; i--) data.push((value >>> i) & 1) }
+  put(4, 4); put(bytes.length, 8); bytes.forEach(b => put(b, 8))
+  for (let i = 0; i < Math.min(4, 55 * 8 - data.length); i++) data.push(0)
+  while (data.length % 8) data.push(0)
+  const codewords = []
+  for (let i = 0; i < data.length; i += 8) codewords.push(data.slice(i, i + 8).reduce((a, b) => (a << 1) | b, 0))
+  for (let p = 0; codewords.length < 55; p++) codewords.push(p % 2 ? 0x11 : 0xec)
+  const exp = Array(512), log = Array(256).fill(0); let x = 1
+  for (let i = 0; i < 255; i++) { exp[i] = x; log[x] = i; x <<= 1; if (x & 0x100) x ^= 0x11d }
+  for (let i = 255; i < 512; i++) exp[i] = exp[i - 255]
+  const mul = (a, b) => a && b ? exp[log[a] + log[b]] : 0
+  let gen = [1]
+  for (let i = 0; i < 15; i++) { const next = Array(gen.length + 1).fill(0); gen.forEach((v, j) => { next[j] ^= v; next[j + 1] ^= mul(v, exp[i]) }); gen = next }
+  const ecc = Array(15).fill(0)
+  codewords.forEach(c => { const factor = c ^ ecc.shift(); ecc.push(0); gen.slice(1).forEach((v, i) => ecc[i] ^= mul(v, factor)) })
+  const bits = [...codewords, ...ecc].flatMap(c => Array.from({ length: 8 }, (_, i) => (c >>> (7 - i)) & 1))
+  const size = 29, m = Array.from({ length: size }, () => Array(size).fill(null))
+  const set = (r, c, v) => { if (r >= 0 && r < size && c >= 0 && c < size) m[r][c] = v ? 1 : 0 }
+  function finder(r, c) { for (let y = -1; y <= 7; y++) for (let x = -1; x <= 7; x++) set(r + y, c + x, y >= 0 && y <= 6 && x >= 0 && x <= 6 && (y === 0 || y === 6 || x === 0 || x === 6 || (y >= 2 && y <= 4 && x >= 2 && x <= 4))) }
+  finder(0, 0); finder(0, 22); finder(22, 0)
+  for (let i = 8; i < 21; i++) { set(6, i, i % 2 === 0); set(i, 6, i % 2 === 0) }
+  for (let y = -2; y <= 2; y++) for (let x = -2; x <= 2; x++) set(22 + y, 22 + x, Math.max(Math.abs(x), Math.abs(y)) !== 1)
+  // Reserve format cells, then write data in the standard zig-zag pattern (mask 0).
+  for (let i = 0; i < 9; i++) { if (i !== 6) { set(8, i, 0); set(i, 8, 0) } }
+  for (let i = 0; i < 8; i++) { set(8, size - 1 - i, 0); set(size - 1 - i, 8, 0) }
+  set(size - 8, 8, 1)
+  let bit = 0, upward = true
